<script>

(function () {

  'use strict';

  /* ============================================================

     重要说明（给加加）

     这套代码里所有 "对象.属性" 都写成了 "对象['属性']"，

     原因是 Pages CMS 会把 xxx.yyy 自动链接化成 [xxx]([http://xxx)，](http://xxx)，)

     导致 <script> 语法崩掉、按钮点不动。

     方括号写法 CMS 碰不到，所以这样写。

     功能、外观、文案和普通写法完全一样。

     ============================================================ */

  /* ============================================================

     一、台词库（想加就往后 push）

     ============================================================ */

  var QUOTES = [

    '呜哇，别捏了啦',

    '咕噜咕噜……好痒',

    '你再捏我就沉下去了',

    '讨厌，人家是软的',

    '啊——变扁了变扁了',

    '捏太重了，会漏水的',

    '咕唧咕唧……听到没',

    '我的脸是不是变形了',

    '轻点轻点，我怕痒',

    '再捏一下就要晕了',

    '抱抱我嘛',

    '陪我待一会儿好不好',

    '摸摸头也可以的',

    '你忙你的，我就看着你',

    '今天有没有多喝水',

    '早点睡，别熬夜',

    '想你了，不过不承认',

    '你笑一个，我看看',

    '我在想海是什么味道',

    '云飘过去了吗',

    '好困，想在水里打滚',

    '世界好安静啊',

    '发呆也是一种本领',

    '你猜我现在在想什么',

    '如果能飞就好了',

    '今天的风是甜的吗',

    '我是海里最圆的那一只',

    '听说捏鲸鱼会变好运',

    '我很好捏的，你看',

    '蓝色最可爱，没错是我',

    '请叫我海洋小汤圆',

    '今天过得还好吗',

    '累了就歇一会儿',

    '没什么大事，就是看看你',

    '你要开心一点',

    '谢谢你来找我玩',

    '明天也要好好的',

    '记得按时吃饭啊',

    '难过的话，捏捏我'

  ];

  /* 初始显示的文字 */

  var INIT_TEXT = '长按我 · 捏一捏 ~';

  /* ============================================================

     二、全局状态

     ============================================================ */

  var bodyEl = null;

  var sayEl = null;

  var lastIndex = -1;

  var isPressed = false;

  /* ============================================================

     三、工具函数

     ============================================================ */

  function pickQuote() {

    if (QUOTES['length'] === 0) return INIT_TEXT;

    if (QUOTES['length'] === 1) return QUOTES[0];

    var i = lastIndex;

    var guard = 0;

    while (i === lastIndex && guard < 20) {

      i = Math['floor'](Math['random']() * QUOTES['length']);

      guard++;

    }

    lastIndex = i;

    return QUOTES[i];

  }

  function showQuote() {

    if (!sayEl) return;

    sayEl['textContent'] = pickQuote();

    // 重新触发淡入动画

    sayEl['classList']['remove']('w-pop');

    // 强制重排，让动画能重新播放

    void sayEl['offsetWidth'];

    sayEl['classList']['add']('w-pop');

  }

  /* ============================================================

     四、按压 / 松开

     ============================================================ */

  function press() {

    if (!bodyEl) return;

    if (isPressed) return;

    isPressed = true;

    bodyEl['classList']['add']('w-squeeze');

  }

  function release() {

    if (!bodyEl) return;

    if (!isPressed) return;

    isPressed = false;

    bodyEl['classList']['remove']('w-squeeze');

    showQuote();

  }

  /* ============================================================

     五、初始化

     ============================================================ */

  function init() {

    var appEl = document['getElementById']('whale-app');

    var _body = document['getElementById']('whale-body');

    var _say  = document['getElementById']('whale-say');

    if (!appEl || !_body || !_say) return false;

    bodyEl = _body;

    sayEl = _say;

    // 初始文字

    sayEl['textContent'] = INIT_TEXT;

    // 指针事件：鼠标和触摸都支持

    bodyEl['addEventListener']('pointerdown', function (e) {

      if (e['button'] !== undefined && e['button'] !== 0) return;

      if (e['preventDefault']) e['preventDefault']();

      press();

    });

    bodyEl['addEventListener']('pointerup', function (e) {

      if (e['preventDefault']) e['preventDefault']();

      release();

    });

    bodyEl['addEventListener']('pointercancel', function () {

      release();

    });

    bodyEl['addEventListener']('pointerleave', function () {

      release();

    });

    // 兜底：如果某些老浏览器不支持 pointer，用 touch / mouse

    if (!window['PointerEvent']) {

      bodyEl['addEventListener']('touchstart', function (e) {

        e['preventDefault']();

        press();

      }, { passive: false });

      bodyEl['addEventListener']('touchend', function (e) {

        e['preventDefault']();

        release();

      }, { passive: false });

      bodyEl['addEventListener']('touchcancel', function () {

        release();

      });

      bodyEl['addEventListener']('mousedown', function (e) {

        e['preventDefault']();

        press();

      });

      bodyEl['addEventListener']('mouseup', function (e) {

        e['preventDefault']();

        release();

      });

      bodyEl['addEventListener']('mouseleave', function () {

        release();

      });

    }

    // 键盘无障碍：空格 / 回车也能捏

    bodyEl['addEventListener']('keydown', function (e) {

      if (e['key'] === ' ' || e['key'] === 'Enter') {

        e['preventDefault']();

        press();

      }

    });

    bodyEl['addEventListener']('keyup', function (e) {

      if (e['key'] === ' ' || e['key'] === 'Enter') {

        e['preventDefault']();

        release();

      }

    });

    // 页面失焦时松开，避免卡在压扁状态

    window['addEventListener']('blur', function () {

      release();

    });

    document['addEventListener']('visibilitychange', function () {

      if (document['hidden']) release();

    });

    return true;

  }

  /* 重试启动：应对 playable 系统把脚本抽到 DOM 之前执行的情况 */

  var tries = 0;

  var MAX_TRIES = 40;

  function tryInit() {

    if (init()) return;

    tries++;

    if (tries < MAX_TRIES) {

      window['setTimeout'](tryInit, 100);

    }

  }

  if (document['readyState'] === 'loading') {

    document['addEventListener']('DOMContentLoaded', tryInit);

  } else {

    tryInit();

  }

})();

</script>
