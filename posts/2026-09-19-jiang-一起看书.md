<script>
(function () {
  'use strict';

  /* ============================================================
     重要说明（给加加）
     这套代码里所有 "对象.属性" 都写成了 "对象['属性']"，
     原因是 Pages CMS 会把 xxx.yyy 自动链接化成 [xxx](http://xxx)，
     导致 <script> 语法崩掉、按钮点不动。
     方括号写法 CMS 碰不到，所以这样写。
     功能、外观、文案和普通写法完全一样。
     ============================================================ */

  /* ============================================================
     一、可调参数
     ============================================================ */
  var CONFIG = {
    wordsPerPage: 500,            // 每页字数
    /* URL 拆开拼接，防止 Pages CMS 把 api.deepseek.com 链接化 */
    apiBase: 'https:' + '//api' + '.deepseek' + '.com/v1',
    apiModel: 'deepseek-chat',
    maxContext: 1500,             // 送给 AI 的正文最多截取多少字
    maxTokens: 1200               // AI 回答的最大 token
  };

  /* localStorage 键名：
     - read_app_api_key：只存 DeepSeek API Key，纯本地，用户点「保存」时写入，清空浏览器数据即清除。
     - read_app_progress：只存阅读进度（书名、分页、当前页、批注），纯本地。
     不存任何账号、密码、隐私文本。 */
  var KEY_API = 'read_app_api_key';
  var KEY_BOOK = 'read_app_progress';

  /* ============================================================
     二、全局状态
     ============================================================ */
  var book = {
    title: '',
    pages: [],
    cur: 0,
    annos: {}    // { 'p0': '批注内容', ... }
  };

  var apiKey = '';

  var appEl, apiKeyEl, apiSaveEl, apiStatusEl;
  var importEl, bookTextEl, loadBtn, uploadBtn, fileInputEl;
  var readerEl, bookNameEl, progressEl, contentEl, pageIndicatorEl;
  var prevBtn, nextBtn;
  var annoPanelEl, annoBodyEl;
  var aiPanelEl, aiBodyEl;
  var menuEl, askBtn, annoBtn;

  var selText = '';

  /* ============================================================
     三、工具函数
     ============================================================ */
  function $(id) { return document['getElementById'](id); }

  function show(el) {
    if (!el) return;
    el['classList']['remove']('r-hidden');
  }

  function hide(el) {
    if (!el) return;
    el['classList']['add']('r-hidden');
  }

  function readKey() {
    try {
      return window['localStorage']['getItem'](KEY_API) || '';
    } catch (e) {
      return '';
    }
  }

  function writeKey(v) {
    try {
      window['localStorage']['setItem'](KEY_API, v);
    } catch (e) { /* 存不了就算了，不影响阅读 */ }
  }

  function readBook() {
    try {
      var s = window['localStorage']['getItem'](KEY_BOOK);
      if (!s) return null;
      return JSON['parse'](s);
    } catch (e) {
      return null;
    }
  }

  function writeBook() {
    try {
      window['localStorage']['setItem'](KEY_BOOK, JSON['stringify'](book));
    } catch (e) { /* 存不了就算了 */ }
  }

  /* ============================================================
     四、分页
     ============================================================ */
  function paginate(text) {
    var clean = text['replace'](/\r\n/g, '\n')['replace'](/\n{3,}/g, '\n\n');
    var pages = [];
    var start = 0;

    while (start < clean['length']) {
      var end = start + CONFIG['wordsPerPage'];
      if (end < clean['length']) {
        // 尽量在段落处断页
        var br = clean['indexOf']('\n\n', end - 80);
        if (br !== -1 && br < end + 120) end = br + 2;
      }
      pages['push'](clean['slice'](start, end)['trim']());
      start = end;
    }

    if (pages['length'] === 0) pages = [''];

    return pages;
  }

  /* ============================================================
     五、渲染阅读区
     ============================================================ */
  function renderPage() {
    if (!book['pages']['length']) return;

    if (book['cur'] < 0) book['cur'] = 0;
    if (book['cur'] >= book['pages']['length']) {
      book['cur'] = book['pages']['length'] - 1;
    }

    bookNameEl['textContent'] = book['title'] || '未命名';

    var p = book['pages'][book['cur']] || '';
    contentEl['textContent'] = p;

    var total = book['pages']['length'];
    var percent = total > 0 ? Math['round']((book['cur'] + 1) / total * 100) : 0;
    progressEl['textContent'] = percent + '%';
    pageIndicatorEl['textContent'] = '第 ' + (book['cur'] + 1) + ' / ' + total + ' 页';

    // 恢复批注显示
    var k = 'p' + book['cur'];
    if (book['annos'][k]) {
      show(annoPanelEl);
      annoBodyEl['textContent'] = book['annos'][k];
    } else {
      hide(annoPanelEl);
      annoBodyEl['textContent'] = '';
    }

    // 换页时收起 AI 面板
    hide(aiPanelEl);
    aiBodyEl['textContent'] = '';

    writeBook();
  }

  /* ============================================================
     六、AI 调用
     ============================================================ */
  function callAI(prompt, onDone, onError) {
    if (!apiKey) {
      onError('请先在上方填写并保存 DeepSeek API Key');
      return;
    }

    /* 截断超长 prompt，防止超出 API 上下文限制 */
    var p = prompt;
    if (p['length'] > CONFIG['maxContext']) {
      p = p['slice'](0, CONFIG['maxContext']) + '…';
    }

    var url = CONFIG['apiBase'] + '/chat/completions';

    var body = {
      model: CONFIG['apiModel'],
      messages: [
        { role: 'system', content: '你是一个温柔、耐心的读书伙伴，回答简洁、真诚，不用客套话。' },
        { role: 'user', content: p }
      ],
      temperature: 0.7,
      max_tokens: CONFIG['maxTokens']
    };

    window['fetch'](url, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': 'Bearer ' + apiKey
      },
      body: JSON['stringify'](body)
    })
    ['then'](function (r) {
      if (!r['ok']) {
        return r['text']()['then'](function (t) {
          throw new Error('API ' + r['status'] + ': ' + t);
        });
      }
      return r['json']();
    })
    ['then'](function (d) {
      var txt = '';
      if (d && d['choices'] && d['choices'][0] && d['choices'][0]['message']) {
        txt = d['choices'][0]['message']['content'] || '';
      }
      onDone(txt);
    })
    ['catch'](function (e) {
      onError(e['message'] || '请求失败');
    });
  }

  /* 问 AI：对选中文字提问 */
  function askAI() {
    if (!selText) return;

    show(aiPanelEl);
    aiBodyEl['innerHTML'] = '<span class="r-panel-loading"></span>正在想……';

    var q =
      '我在读《' + (book['title'] || '未命名') + '》，选中了这段文字：\n\n' +
      '「' + selText + '」\n\n' +
      '请帮我解释这段话的意思，或者说说你的理解。';

    callAI(q, function (txt) {
      aiBodyEl['textContent'] = txt || '（没有返回内容）';
    }, function (err) {
      aiBodyEl['textContent'] = err;
    });
  }

  /* 写批注：对选中文字生成批注并保存 */
  function writeAnno() {
    if (!selText) return;

    show(aiPanelEl);
    aiBodyEl['innerHTML'] = '<span class="r-panel-loading"></span>正在写批注……';

    var q =
      '请为下面这段书里的文字写一段简短的批注，像读书伙伴在书页边写的感想，100 字以内，语气温暖：\n\n' +
      '「' + selText + '」';

    callAI(q, function (txt) {
      var note = txt || '（没有返回内容）';
      aiBodyEl['textContent'] = note;

      var k = 'p' + book['cur'];
      var old = book['annos'][k] || '';
      book['annos'][k] = old ? (old + '\n\n---\n\n' + note) : note;
      writeBook();

      show(annoPanelEl);
      annoBodyEl['textContent'] = book['annos'][k];
    }, function (err) {
      aiBodyEl['textContent'] = err;
    });
  }

  /* ============================================================
     七、选中文字处理
     ============================================================ */
  function onSelect() {
    var sel = window['getSelection']();
    if (!sel) return;

    var txt = sel['toString']()['trim']();
    if (!txt || !contentEl) {
      menuEl['classList']['remove']('show');
      selText = '';
      return;
    }

    // 判断选区是否在正文里
    var node = sel['anchorNode'];
    if (!node || !contentEl['contains'](node)) {
      menuEl['classList']['remove']('show');
      selText = '';
      return;
    }

    selText = txt;

    // 菜单定位到选区上方
    try {
      var rect = sel['getRangeAt'](0)['getBoundingClientRect']();
      var appRect = appEl['getBoundingClientRect']();

      var left = rect['left'] - appRect['left'] + rect['width'] / 2 - 70;
      var top = rect['top'] - appRect['top'] - 48;

      if (left < 8) left = 8;
      if (top < 8) top = rect['bottom'] - appRect['top'] + 8;

      menuEl['style']['left'] = left + 'px';
      menuEl['style']['top'] = top + 'px';
    } catch (e) { /* 老浏览器忽略定位 */ }

    menuEl['classList']['add']('show');
  }

  function onDocMouseDown(e) {
    if (!menuEl) return;
    if (menuEl['contains'](e['target'])) return;
    menuEl['classList']['remove']('show');
  }

  /* ============================================================
     八、加载书籍
     ============================================================ */
  function startReading(text, title) {
    var t = (text || '')['trim']();
    if (!t) {
      // 不用 alert，用状态栏提示
      apiStatusEl['textContent'] = '请先粘贴或上传内容';
      return;
    }

    book = {
      title: title || '未命名',
      pages: paginate(t),
      cur: 0,
      annos: {}
    };

    writeBook();

    hide(importEl);
    show(readerEl);

    renderPage();
  }

  /* ============================================================
     九、初始化
     ============================================================ */
  function init() {
    var _app = $('read-app');
    var _apiKey = $('r-api-key');
    var _apiSave = $('r-api-save');
    var _apiStatus = $('r-api-status');

    var _import = $('r-import');
    var _bookText = $('r-book-text');
    var _load = $('r-load');
    var _upload = $('r-upload');
    var _file = $('r-file');

    var _reader = $('r-reader');
    var _bookName = $('r-book-name');
    var _progress = $('r-progress');
    var _content = $('r-content');
    var _pageIndicator = $('r-page-indicator');
    var _prev = $('r-prev');
    var _next = $('r-next');

    var _annoPanel = $('r-anno-panel');
    var _annoBody = $('r-anno-body');
    var _aiPanel = $('r-ai-panel');
    var _aiBody = $('r-ai-body');

    var _menu = $('r-menu');
    var _ask = $('r-ask');
    var _anno = $('r-anno');

    if (!_app || !_apiKey || !_load || !_content || !_menu) return false;

    appEl = _app;
    apiKeyEl = _apiKey;
    apiSaveEl = _apiSave;
    apiStatusEl = _apiStatus;

    importEl = _import;
    bookTextEl = _bookText;
    loadBtn = _load;
    uploadBtn = _upload;
    fileInputEl = _file;

    readerEl = _reader;
    bookNameEl = _bookName;
    progressEl = _progress;
    contentEl = _content;
    pageIndicatorEl = _pageIndicator;
    prevBtn = _prev;
    nextBtn = _next;

    annoPanelEl = _annoPanel;
    annoBodyEl = _annoBody;
    aiPanelEl = _aiPanel;
    aiBodyEl = _aiBody;

    menuEl = _menu;
    askBtn = _ask;
    annoBtn = _anno;

    /* ---------- API Key ---------- */
    apiKey = readKey();
    if (apiKey) {
      apiKeyEl['value'] = apiKey;
      apiStatusEl['textContent'] = '已保存';
    } else {
      apiStatusEl['textContent'] = '未配置';
    }

    apiSaveEl['addEventListener']('click', function () {
      var v = apiKeyEl['value']['trim']();
      if (!v) {
        apiStatusEl['textContent'] = 'Key 不能为空';
        return;
      }
      apiKey = v;
      writeKey(v);
      apiStatusEl['textContent'] = '已保存';
    });

    /* ---------- 加载书 ---------- */
    loadBtn['addEventListener']('click', function () {
      startReading(bookTextEl['value'], '未命名');
    });

    uploadBtn['addEventListener']('click', function () {
      if (fileInputEl) fileInputEl['click']();
    });

    if (fileInputEl) {
      fileInputEl['addEventListener']('change', function () {
        var f = fileInputEl['files'][0];
        if (!f) return;

        var reader = new FileReader();
        reader['onload'] = function (e) {
          var title = f['name']['replace'](/\.txt$/i, '');
          startReading(e['target']['result'], title);
        };
        reader['readAsText'](f);
        fileInputEl['value'] = '';
      });
    }

    /* ---------- 翻页 ---------- */
    prevBtn['addEventListener']('click', function () {
      if (book['cur'] > 0) {
        book['cur']--;
        renderPage();
      }
    });

    nextBtn['addEventListener']('click', function () {
      if (book['cur'] < book['pages']['length'] - 1) {
        book['cur']++;
        renderPage();
      }
    });

    /* ---------- 选中文字 ---------- */
    document['addEventListener']('mouseup', onSelect);
    document['addEventListener']('touchend', function () {
      // 触摸结束后稍等一下，等选区稳定
      window['setTimeout'](onSelect, 60);
    });
    document['addEventListener']('mousedown', onDocMouseDown);
    document['addEventListener']('touchstart', function (e) {
      if (!menuEl['contains'](e['target'])) menuEl['classList']['remove']('show');
    }, { passive: true });

    /* ---------- 菜单按钮 ---------- */
    askBtn['addEventListener']('click', function () {
      menuEl['classList']['remove']('show');
      askAI();
    });

    annoBtn['addEventListener']('click', function () {
      menuEl['classList']['remove']('show');
      writeAnno();
    });

    /* ---------- 恢复上次阅读 ---------- */
    var saved = readBook();
    if (saved && saved['pages'] && saved['pages']['length']) {
      book = saved;
      hide(importEl);
      show(readerEl);
      renderPage();
    }

    return true;
  }

  /* 重试启动：应对 playable 系统把脚本抽到 DOM 之前执行的情况 */
  var tries = 0;
  var MAX_TRIES = 40;

  function tryInit() {
    if (init()) return;
    tries++;
    if (tries < MAX_TRIES) {
      window['setTimeout'](tryInit, 100);
    }
  }

  if (document['readyState'] === 'loading') {
    document['addEventListener']('DOMContentLoaded', tryInit);
  } else {
    tryInit();
  }
})();
</script>
