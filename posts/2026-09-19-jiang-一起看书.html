function buildHuffmanTable(codeLengths) {

var n = codeLengths['length'];

var maxBits = 0;

}

out['push'](sym);

} else {

var lenIdx = sym - 257;

if (lenIdx >= 29) break;

var len = lengthBase[lenIdx];

var extraLen = lengthExtra[lenIdx];

if (extraLen > 0) len += reader['readBits'](extraLen);

var distSym = decodeSymbol(reader, distHuff);

var dist = distBase[distSym];

var extraDist = distExtra[distSym];

if (extraDist > 0) dist += reader['readBits'](extraDist);

var start = out['length'] - dist;

for (var i = 0; i < len; i++) {

out['push'](out[start + i]);

}

}

}

}

function inflateRaw(data) {

var reader = new BitReader(data);

var out = [];

while (true) {

var bfinal = reader['readBit']();

var btype = reader['readBits'](2);

if (btype === 0) {

reader['skipToByte']();

var len = reader['data'][reader['pos']] | (reader['data'][reader['pos'] + 1] << 8);

var nlen = reader['data'][reader['pos'] + 2] | (reader['data'][reader['pos'] + 3] << 8);

if ((len ^ nlen ^ 0xffff) !== 0) break;

reader['pos'] += 4;

for (var i = 0; i < len; i++) out['push'](reader['data'][reader['pos']++]);

} else if (btype === 1) {

inflateBlock(reader, buildHuffmanTable(fixedLiteralCodeLengths()), buildHuffmanTable(fixedDistanceCodeLengths()), out);

} else if (btype === 2) {

var hlit = reader['readBits'](5) + 257;

var hdist = reader['readBits'](5) + 1;

var hclen = reader['readBits'](4) + 4;

var clCodes = [16,17,18,0,8,7,9,6,10,5,11,4,12,3,13,2,14,1,15];

var clenLen = new Array(hclen);

for (var j = 0; j < hclen; j++) clenLen[clCodes[j]] = reader['readBits'](3);

var clHuff = buildHuffmanTable(clenLen);

var total = hlit + hdist;

var allLen = new Array(total);

var idx = 0;

while (idx < total) {

var sym = decodeSymbol(reader, clHuff);

if (sym < 16) {

allLen[idx++] = sym;

} else if (sym === 16) {

var rep = reader['readBits'](2) + 3;

var prev = idx > 0 ? allLen[idx - 1] : 0;

for (var k = 0; k < rep && idx < total; k++) allLen[idx++] = prev;

} else if (sym === 17) {

var rep2 = reader['readBits'](3) + 3;

for (var m = 0; m < rep2 && idx < total; m++) allLen[idx++] = 0;

} else if (sym === 18) {

var rep3 = reader['readBits'](7) + 11;

for (var n = 0; n < rep3 && idx < total; n++) allLen[idx++] = 0;

} else break;

}

var litHuff = buildHuffmanTable(allLen['slice'](0, hlit));

var distHuff = buildHuffmanTable(allLen['slice'](hlit, total));

inflateBlock(reader, litHuff, distHuff, out);

} else { break; }

if (bfinal) break;

}

return new Uint8Array(out);

}

function scanZip(arrayBuffer) {

var view = new DataView(arrayBuffer);

var bytes = new Uint8Array(arrayBuffer);

var eocdOffset = -1;

var minSearch = Math['max'](0, arrayBuffer['byteLength'] - 65557);

for (var i = arrayBuffer['byteLength'] - 22; i >= minSearch; i--) {

if (view['getUint32'](i, true) === 0x06054b50) { eocdOffset = i; break; }

}

if (eocdOffset === -1) throw new Error('不是有效的 EPUB 文件');

var cdOffset = view['getUint32'](eocdOffset + 16, true);

var cdCount = view['getUint16'](eocdOffset + 10, true);

var entriesByPath = {};

var offset = cdOffset;

for (var n = 0; n < cdCount; n++) {

if (view['getUint32'](offset, true) !== 0x02014b50) break;

var compression = view['getUint16'](offset + 10, true);

var compressedSize = view['getUint32'](offset + 20, true);

var fileNameLength = view['getUint16'](offset + 28, true);

var extraLength = view['getUint16'](offset + 30, true);

var commentLength = view['getUint16'](offset + 32, true);

var localHeaderOffset = view['getUint32'](offset + 42, true);

var nameBytes = bytes['slice'](offset + 46, offset + 46 + fileNameLength);

var name = new TextDecoder('utf-8')['decode'](nameBytes);

entriesByPath[name] = { name: name, compression: compression, compressedSize: compressedSize, localHeaderOffset: localHeaderOffset };

offset += 46 + fileNameLength + extraLength + commentLength;

}

return { view: view, bytes: bytes, entries: entriesByPath };

}

function decompressSync(scan, entry) {

var view = scan['view'];

var bytes = scan['bytes'];

var lnl = view['getUint16'](entry['localHeaderOffset'] + 26, true);

var lel = view['getUint16'](entry['localHeaderOffset'] + 28, true);

var dataStart = entry['localHeaderOffset'] + 30 + lnl + lel;

/* 检查 general purpose bit flag 第 3 位：若为 1，数据后有 16 字节 data descriptor */

var gpbf = view['getUint16'](entry['localHeaderOffset'] + 6, true);

var hasDD = (gpbf & 0x08) !== 0;

var compSize = entry['compressedSize'];

if (hasDD) compSize -= 16; /* 跳过 descriptor 后才是真实压缩数据 */

var cd = bytes['slice'](dataStart, dataStart + compSize);

if (entry['compression'] === 0) return cd;

if (entry['compression'] === 8) { try { return inflateRaw(cd); } catch (e) { return null; } }

return null;

}

function parseEpub(file, onProgress) {

var decoder = new TextDecoder('utf-8');

var parser = new DOMParser();

function progress(s, c, t) { if (typeof onProgress === 'function') { try { onProgress(s, c, t); } catch (e) { } } }

return file['arrayBuffer']()['then'](function (buf) {

progress('扫描目录', 1, 1);

var scan = scanZip(buf);

progress('解压 container.xml', 1, 1);

var ce = scan['entries']['META-INF/container.xml'];

if (!ce) throw new Error('EPUB 结构异常：找不到 container.xml');

var cb = decompressSync(scan, ce);

if (!cb) throw new Error('EPUB 结构异常：container.xml 解压失败');

var cd = parser['parseFromString'](decoder['decode'](cb), 'application/xml');

var rf = cd['querySelector']('rootfile');

if (!rf) throw new Error('EPUB 结构异常：找不到 rootfile');

var opfPath = rf['getAttribute']('full-path');

var slash = opfPath['lastIndexOf']('/');

var baseDir = slash !== -1 ? opfPath['substring'](0, slash + 1) : '';

progress('解压 OPF', 1, 1);

var oe = scan['entries'][opfPath];

if (!oe) throw new Error('EPUB 结构异常：找不到 OPF');

var ob = decompressSync(scan, oe);

if (!ob) throw new Error('OPF 解压失败');

var od = parser['parseFromString'](decoder['decode'](ob), 'application/xml');

var manifest = {};

var items = od['querySelectorAll']('manifest > item');

for (var i = 0; i < items['length']; i++) {

var id = items[i]['getAttribute']('id');

var href = items[i]['getAttribute']('href');

if (id && href) manifest[id] = href;

}

var spine = [];

var refs = od['querySelectorAll']('spine > itemref');

for (var j = 0; j < refs['length']; j++) {

var idref = refs[j]['getAttribute']('idref');

if (idref) spine['push'](idref);

}

var titleEl = od['getElementsByTagNameNS']('http://purl.org/dc/elements/1.1/', 'title')[0];

var title = titleEl ? titleEl['textContent']['trim']() : '';

var cps = [];

for (var k = 0; k < spine['length']; k++) {

var href2 = manifest[spine[k]];

if (!href2) continue;

var rel = baseDir + decodeURIComponent(href2);

var entry = scan['entries'][rel];

if (entry) cps['push']({ rel: rel, entry: entry });

}

progress('解压章节', 0, cps['length']);

var chapters = new Array(cps['length']);

for (var t = 0; t < cps['length']; t++) {

chapters[t] = decompressSync(scan, cps[t]['entry']);

progress('解压章节', t + 1, cps['length']);

}

progress('提取文字', 0, cps['length']);

var fullText = '';

for (var m = 0; m < chapters['length']; m++) {

if (!chapters[m]) { progress('提取文字', m + 1, cps['length']); continue; }

var xmlStr = decoder['decode'](chapters[m]);

var ch;

try { ch = parser['parseFromString'](xmlStr, 'application/xml'); if (ch['querySelector']('parsererror')) throw new Error('x'); }

catch (e) { ch = parser['parseFromString'](xmlStr, 'text/html'); }

var junk = ch['querySelectorAll']('script, style');

for (var p = 0; p < junk['length']; p++) junk[p]['remove']();

var txt = ch['body'] ? ch['body']['textContent'] : '';

txt = txt['replace'](/\s+\n/g, '\n')['replace'](/\n{3,}/g, '\n\n')['trim']();

if (txt) fullText += txt + '\n\n';

progress('提取文字', m + 1, cps['length']);

}

return { title: title, text: fullText['trim']() };

});

}

return { parse: parseEpub };

}
