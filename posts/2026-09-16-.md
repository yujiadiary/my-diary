(function () {

  'use strict';

  /* ============================================================

     可调参数区：想改数值直接改这里

     ============================================================ */

  var CONFIG = {

    cols: 15,

    rows: 15,

    maxBoardPx: 450,

    baseSpeed: 175,

    minSpeed: 85,

    speedStep: 4,

    startLen: 3,

    cornerRatio: 0.3,

    insetRatio: 0.06

  };

  var COLOR = {

    boardBg: '#FFFDFB',

    boardAlt: '#FBF1F6',

    snakeHead: '#7ED6B5',

    snakeTail: '#C7CEEA',

    food: '#FF9FB2',

    eye: '#4A4453',

    blush: '#FFB7C5'

  };

  var TEXT = {

    idleTitle: '准备好了吗',

    idleText: '点「开始」，喂小蛇吃爱心',

    pauseTitle: '暂停中',

    pauseText: '点「继续」接着玩',

    overTitle: '撞到啦',

    overEmpty: '一颗爱心都没吃到，再试一次吧',

    overScore: function (n) { return '这局吃了 ' + n + ' 颗爱心'; },

    newBest: '　新纪录！'

  };

  // localStorage 键名：只存历史最高分（一个整数），纯本地。

  // 不存任何账号、密码、隐私文本。清空浏览器数据即被清除。

  var BEST_KEY = 'macaron-snake-best-v1';

  var canvas = document.getElementById('board');

  var ctx = canvas.getContext('2d');

  var stage = document.getElementById('stage');

  var overlay = document.getElementById('overlay');

  var ovTitle = document.getElementById('ovTitle');

  var ovText = document.getElementById('ovText');

  var scoreEl = document.getElementById('score');

  var bestEl = document.getElementById('best');

  var btnMain = document.getElementById('btnMain');

  var btnReset = document.getElementById('btnReset');

  var pad = document.getElementById('pad');

  var DIRS = {

    up:    { x: 0,  y: -1 },

    down:  { x: 0,  y: 1  },

    left:  { x: -1, y: 0  },

    right: { x: 1,  y: 0  }

  };

  var cell = 24;

  var snake = [];

  var dir = DIRS.right;

  var pendingDirs = [];

  var food = { x: 0, y: 0 };

  var score = 0;

  var best = readBest();

  var newBest = false;

  var timer = null;

  var state = 'idle';

  function readBest() {

    try {

      var v = parseInt(window.localStorage.getItem(BEST_KEY), 10);

      return isNaN(v) || v < 0 ? 0 : v;

    } catch (e) {

      return 0;

    }

  }

  function writeBest(v) {

    try {

      window.localStorage.setItem(BEST_KEY, String(v));

    } catch (e) { /* 存不了就算了 */ }

  }

  function hexToRgb(hex) {

    var v = parseInt(hex.slice(1), 16);

    return { r: (v >> 16) & 255, g: (v >> 8) & 255, b: v & 255 };

  }

  function lerpColor(a, b, t) {

    var A = hexToRgb(a), B = hexToRgb(b);

    var r = Math.round(A.r + (B.r - A.r) * t);

    var g = Math.round(A.g + (B.g - A.g) * t);

    var bl = Math.round(A.b + (B.b - A.b) * t);

    return 'rgb(' + r + ',' + g + ',' + bl + ')';

  }

  function roundRect(x, y, w, h, r) {

    var rr = Math.min(r, w / 2, h / 2);

    ctx.beginPath();

    ctx.moveTo(x + rr, y);

    ctx.arcTo(x + w, y, x + w, y + h, rr);

    ctx.arcTo(x + w, y + h, x, y + h, rr);

    ctx.arcTo(x, y + h, x, y, rr);

    ctx.arcTo(x, y, x + w, y, rr);

    ctx.closePath();

  }

  function fit() {

    var cs = window.getComputedStyle(stage);

    var avail = stage.clientWidth

              - parseFloat(cs.paddingLeft)

              - parseFloat(cs.paddingRight);

    if (!avail || avail < 0) avail = 300;

    var size = Math.min(avail, CONFIG.maxBoardPx);

    cell = Math.max(8, Math.floor(size / CONFIG.cols));

    var px = cell * CONFIG.cols;

    [canvas.style](http://canvas.style).width = px + 'px';

    [canvas.style](http://canvas.style).height = px + 'px';

    var dpr = window.devicePixelRatio || 1;

    canvas.width = Math.round(px * dpr);

    canvas.height = Math.round(px * dpr);

    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);

    draw();

  }

  function reset() {

    stopTimer();

    state = 'idle';

    newBest = false;

    score = 0;

    dir = DIRS.right;

    pendingDirs = [];

    snake = [];

    var hy = Math.floor(CONFIG.rows / 2);

    var hx = Math.floor(CONFIG.cols / 2);

    for (var i = 0; i < CONFIG.startLen; i++) {

      snake.push({ x: hx - i, y: hy });

    }

    placeFood();

    renderHUD();

    syncUI();

    draw();

  }

  function placeFood() {

    var free = [];

    for (var y = 0; y < CONFIG.rows; y++) {

      for (var x = 0; x < CONFIG.cols; x++) {

        var hit = false;

        for (var i = 0; i < snake.length; i++) {

          if (snake[i].x === x && snake[i].y === y) { hit = true; break; }

        }

        if (!hit) free.push({ x: x, y: y });

      }

    }

    if (free.length === 0) {

      gameOver(true);

      return;

    }

    food = free[Math.floor(Math.random() * free.length)];

  }

  function currentSpeed() {

    return Math.max(CONFIG.minSpeed, CONFIG.baseSpeed - score * CONFIG.speedStep);

  }

  function startTimer() {

    stopTimer();

    timer = window.setInterval(tick, currentSpeed());

  }

  function stopTimer() {

    if (timer !== null) {

      window.clearInterval(timer);

      timer = null;

    }

  }

  function tick() {

    while (pendingDirs.length > 0) {

      var d = pendingDirs.shift();

      if (d.x === -dir.x && d.y === -dir.y) continue;

      if (d.x === dir.x && d.y === dir.y) continue;

      dir = d;

      break;

    }

    var head = { x: snake[0].x + dir.x, y: snake[0].y + dir.y };

    if (head.x < 0 || head.y < 0 ||

        head.x >= CONFIG.cols || head.y >= CONFIG.rows) {

      gameOver(false);

      return;

    }

    var willGrow = (head.x === food.x && head.y === food.y);

    var bodyEnd = willGrow ? snake.length : snake.length - 1;

    for (var i = 0; i < bodyEnd; i++) {

      if (snake[i].x === head.x && snake[i].y === head.y) {

        gameOver(false);

        return;

      }

    }

    snake.unshift(head);

    if (willGrow) {

      score++;

      renderHUD();

      placeFood();

      startTimer();

      if (state === 'over') return;

    } else {

      snake.pop();

    }

    draw();

  }

  function gameOver(isFull) {

    stopTimer();

    state = 'over';

    newBest = score > best;

    if (newBest) {

      best = score;

      writeBest(best);

    }

    renderHUD();

    syncUI();

    draw();

    void isFull;

  }

  function start() {

    if (state === 'running' || state === 'over') return;

    state = 'running';

    startTimer();

    syncUI();

    draw();

  }

  function pause() {

    if (state !== 'running') return;

    state = 'paused';

    stopTimer();

    syncUI();

    draw();

  }

  function onMainButton() {

    if (state === 'idle' || state === 'paused') {

      start();

    } else if (state === 'running') {

      pause();

    } else if (state === 'over') {

      reset();

      start();

    }

  }

  function renderHUD() {

    scoreEl.textContent = String(score);

    bestEl.textContent = String(best);

  }

  function syncUI() {

    if (state === 'running') {

      overlay.classList.add('hide');

      btnMain.textContent = '暂停';

      return;

    }

    overlay.classList.remove('hide');

    if (state === 'idle') {

      ovTitle.textContent = TEXT.idleTitle;

      ovText.textContent = TEXT.idleText;

      btnMain.textContent = '开始';

    } else if (state === 'paused') {

      ovTitle.textContent = TEXT.pauseTitle;

      ovText.textContent = TEXT.pauseText;

      btnMain.textContent = '继续';

    } else if (state === 'over') {

      ovTitle.textContent = TEXT.overTitle;

      var msg = score > 0 ? TEXT.overScore(score) : TEXT.overEmpty;

      if (newBest && score > 0) msg += TEXT.newBest;

      ovText.textContent = msg;

      btnMain.textContent = '再来一局';

    }

  }

  function draw() {

    if (!snake || snake.length === 0) return;

    var W = CONFIG.cols * cell;

    var H = CONFIG.rows * cell;

    ctx.clearRect(0, 0, W, H);

    ctx.fillStyle = COLOR.boardBg;

    ctx.fillRect(0, 0, W, H);

    ctx.fillStyle = COLOR.boardAlt;

    for (var y = 0; y < CONFIG.rows; y++) {

      for (var x = 0; x < CONFIG.cols; x++) {

        if ((x + y) % 2 === 0) {

          ctx.fillRect(x  *cell, y*  cell, cell, cell);

        }

      }

    }

    drawFood();

    drawSnake();

  }

  function drawFood() {

    var cx = food.x * cell + cell / 2;

    var cy = food.y * cell + cell / 2;

    var s = cell * 0.28;

    ctx.fillStyle = [COLOR.food](http://COLOR.food);

    ctx.beginPath();

    ctx.moveTo(cx, cy + s);

    ctx.bezierCurveTo(cx - s  *1.7, cy - s*  0.45,

                      cx - s  *0.65, cy - s*  1.45, cx, cy - s * 0.45);

    ctx.bezierCurveTo(cx + s  *0.65, cy - s*  1.45,

                      cx + s  *1.7, cy - s*  0.45, cx, cy + s);

    ctx.closePath();

    ctx.fill();

    ctx.fillStyle = 'rgba(255, 255, 255, 0.72)';

    ctx.beginPath();

    ctx.arc(cx - s  *0.5, cy - s*  0.5, Math.max(1, s  *0.24), 0, Math.PI*  2);

    ctx.fill();

  }

  function drawSnake() {

    var n = snake.length;

    var inset = Math.max(1, cell * CONFIG.insetRatio);

    var baseR = cell * CONFIG.cornerRatio;

    for (var i = n - 1; i >= 0; i--) {

      var seg = snake[i];

      var t = n === 1 ? 0 : i / (n - 1);

      var x = seg.x * cell + inset;

      var y = seg.y * cell + inset;

      var w = cell - inset * 2;

      ctx.fillStyle = lerpColor(COLOR.snakeHead, COLOR.snakeTail, t);

      roundRect(x, y, w, w, i === 0 ? baseR * 1.25 : baseR);

      ctx.fill();

    }

    drawFace();

  }

  function drawFace() {

    var h = snake[0];

    if (!h) return;

    var cx = h.x * cell + cell / 2;

    var cy = h.y * cell + cell / 2;

    var ex = dir.x  *cell*  0.12;

    var ey = dir.y  *cell*  0.12;

    var perpX = -dir.y;

    var perpY = dir.x;

    var off = cell * 0.2;

    var eyeR = Math.max(1.4, cell * 0.085);

    ctx.fillStyle = COLOR.eye;

    ctx.beginPath();

    ctx.arc(cx + ex + perpX  *off, cy + ey + perpY*  off, eyeR, 0, Math.PI * 2);

    ctx.fill();

    ctx.beginPath();

    ctx.arc(cx + ex - perpX  *off, cy + ey - perpY*  off, eyeR, 0, Math.PI * 2);

    ctx.fill();

    var blushR = Math.max(1.2, cell * 0.075);

    var bx = cx - dir.x  *cell*  0.26;

    var by = cy - dir.y  *cell*  0.26;

    var boff = cell * 0.3;

    ctx.fillStyle = COLOR.blush;

    ctx.globalAlpha = 0.85;

    ctx.beginPath();

    ctx.arc(bx + perpX  *boff, by + perpY*  boff, blushR, 0, Math.PI * 2);

    ctx.fill();

    ctx.beginPath();

    ctx.arc(bx - perpX  *boff, by - perpY*  boff, blushR, 0, Math.PI * 2);

    ctx.fill();

    ctx.globalAlpha = 1;

  }

  function turn(name) {

    var d = DIRS[name];

    if (!d) return;

    if (state !== 'running' && state !== 'paused') return;

    if (pendingDirs.length >= 3) return;

    pendingDirs.push(d);

  }

  btnMain.addEventListener('click', onMainButton);

  btnReset.addEventListener('click', function () {

    reset();

  });

  pad.addEventListener('click', function (e) {

    var btn = [e.target](http://e.target).closest('.pad-btn');

    if (!btn) return;

    turn(btn.getAttribute('data-dir'));

  });

  document.addEventListener('keydown', function (e) {

    var key = e.key;

    var tag = [e.target](http://e.target) && [e.target](http://e.target).tagName;

    if (key === 'ArrowUp' || key === 'w' || key === 'W') {

      turn('up'); e.preventDefault();

    } else if (key === 'ArrowDown' || key === 's' || key === 'S') {

      turn('down'); e.preventDefault();

    } else if (key === 'ArrowLeft' || key === 'a' || key === 'A') {

      turn('left'); e.preventDefault();

    } else if (key === 'ArrowRight' || key === 'd' || key === 'D') {

      turn('right'); e.preventDefault();

    } else if (key === ' ' || key === 'Spacebar') {

      if (tag === 'BUTTON') return;

      e.preventDefault();

      onMainButton();

    }

  });

  var touchStart = null;

  canvas.addEventListener('touchstart', function (e) {

    var t = e.changedTouches[0];

    touchStart = { x: t.clientX, y: t.clientY };

  }, { passive: true });

  canvas.addEventListener('touchend', function (e) {

    if (!touchStart) return;

    var t = e.changedTouches[0];

    var dx = t.clientX - touchStart.x;

    var dy = t.clientY - touchStart.y;

    touchStart = null;

    if (Math.abs(dx) < 22 && Math.abs(dy) < 22) return;

    if (Math.abs(dx) > Math.abs(dy)) {

      turn(dx > 0 ? 'right' : 'left');

    } else {

      turn(dy > 0 ? 'down' : 'up');

    }

  }, { passive: true });

  window.addEventListener('blur', function () {

    if (state === 'running') pause();

  });

  window.addEventListener('resize', function () {

    fit();

  });

  fit();

  reset();

  fit();

})();
